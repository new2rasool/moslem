# Python 3.9 - 3.11 recommended (py-tgcalls/ntgcalls wheels)

# ---- MTProto client (ONLY Pyrogram - no Telethon mixed in) ----
pyrogram==2.0.106

# ---- Voice/Video call engine (classic API: join_group_call) ----
# py-tgcalls 1.2.9 is built against ntgcalls 1.1.x (title-case enums
# StreamStatus.Playing / InputMode.Shell). ntgcalls==1.1.3 ships
# prebuilt wheels for Linux/macOS/Windows on Python 3.8 - 3.12.
# (clients.py contains a small shim that also tolerates ntgcalls 1.2.x)
py-tgcalls==1.2.9
ntgcalls==1.1.3

# ---- Downloads / YouTube ----
# yt-dlp is the primary engine for every YouTube command: download (with a
# bv*+ba format fallback chain, merged by ffmpeg) AND search (`ytsearch`).
yt-dlp==2026.7.4
# Optional fallback for YouTube search only. It scrapes the results HTML and
# is currently broken against modern requests/urllib3 - calling
# VideosSearch(...).result() raises:
#   TypeError: post() got an unexpected keyword argument 'proxies'
# yt-dlp search runs first, so a failure here is non-fatal.
youtube-search-python==1.6.6

# ---- Persian (Jalali) calendar for timestamps ----
jdatetime==6.0.1

# ---- Async HTTP (used by py-tgcalls and Melobit search) ----
aiohttp==3.9.5

# ---- py-tgcalls runtime dependencies ----
psutil==5.9.8
screeninfo==0.8.1
deprecation==2.1.0
